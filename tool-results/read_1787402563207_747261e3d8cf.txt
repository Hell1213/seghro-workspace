   107→export function DashboardSection() {
   108→  const ref = useRef<HTMLDivElement>(null);
   109→  const isInView = useInView(ref, { once: true, margin: '-100px' });
   110→  const { data: session } = useSession();
   111→  const [activeTab, setActiveTab] = useState<TabId>('overview');
   112→
   113→  // Auth-aware banner: demo mode vs authenticated user
   114→  const isAuthenticated = !!session?.user?.email;
   115→  const [bannerDismissed, setBannerDismissed] = useState(false);
   116→
   117→  useEffect(() => {
   118→    try {
   119→      const dismissed = localStorage.getItem('sentinel-auth-banner-dismissed');
   120→      if (dismissed === 'true') setBannerDismissed(true);
   121→    } catch { /* ignore */ }
   122→  }, []);
   123→
   124→  const handleDismissBanner = useCallback(() => {
   125→    setBannerDismissed(true);
   126→    try {
   127→      localStorage.setItem('sentinel-auth-banner-dismissed', 'true');
   128→    } catch { /* ignore */ }
   129→  }, []);
   130→
   131→  const showBanner = useMemo(
   132→    () => !bannerDismissed,
   133→    [bannerDismissed]
   134→  );
   135→
   136→  // Sync tab to URL hash
   137→  useEffect(() => {
   138→    const hash = window.location.hash.replace('#dashboard-', '').replace('#', '');
   139→    if (['overview', 'traces', 'issues', 'alerts', 'api-health'].includes(hash)) {
   140→      setActiveTab(hash as TabId);
   141→    }
   142→  }, []);
   143→
   144→  const handleTabChange = (tab: TabId) => {
   145→    setActiveTab(tab);
   146→    window.history.replaceState(null, '', `#dashboard-${tab}`);
   147→  };
   148→  const [selectedAgentId, setSelectedAgentId] = useState<string | null>(null);
   149→  const [detailAgent, setDetailAgent] = useState<Agent | null>(null);
   150→  const [comparisonIds, setComparisonIds] = useState<string[]>([]);
   151→  const [comparisonAgents, setComparisonAgents] = useState<[Agent, Agent] | null>(null);
   152→  const [showWaterfall, setShowWaterfall] = useState(true);
   153→  const [settingsOpen, setSettingsOpen] = useState(false);
   154→  const [createAgentOpen, setCreateAgentOpen] = useState(false);
   155→  const [simulateTraceOpen, setSimulateTraceOpen] = useState(false);
   156→  const [agents, setAgents] = useState<Agent[]>([]);
   157→  const [traces, setTraces] = useState<Trace[]>([]);
   158→  const [issues, setIssues] = useState<Issue[]>([]);
   159→  const [alertItems, setAlertItems] = useState<AlertItem[]>([]);
   160→  const [metrics, setMetrics] = useState<any>(null);
   161→  const [loading, setLoading] = useState(true);
   162→
   163→  // Search and filter state
   164→  const [issueSearch, setIssueSearch] = useState('');
   165→  const [issueSeverityFilter, setIssueSeverityFilter] = useState<string>('all');
   166→  const [issueStatusFilter, setIssueStatusFilter] = useState<string>('all');
   167→  const [showFilters, setShowFilters] = useState(false);
   168→
   169→  // Trace search and filter state
   170→  const [traceSearch, setTraceSearch] = useState('');
   171→  const [traceStatusFilter, setTraceStatusFilter] = useState<string>('all');
   172→
   173→  // Filter persistence via localStorage
   174→  useEffect(() => {
   175→    try {
   176→      const saved = localStorage.getItem('sentinel-dash-filters');
   177→      if (saved) {
   178→        const f = JSON.parse(saved);
   179→        if (f.traceSearch) setTraceSearch(f.traceSearch);
   180→        if (f.issueSearch) setIssueSearch(f.issueSearch);
   181→        if (f.traceStatusFilter) setTraceStatusFilter(f.traceStatusFilter);
   182→        if (f.issueSeverityFilter) setIssueSeverityFilter(f.issueSeverityFilter);
   183→        if (f.issueStatusFilter) setIssueStatusFilter(f.issueStatusFilter);
   184→      }
   185→    } catch { /* ignore */ }
   186→  }, []);
   187→
   188→  useEffect(() => {
   189→    try {
   190→      localStorage.setItem('sentinel-dash-filters', JSON.stringify({
   191→        traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter,
   192→      }));
   193→    } catch { /* ignore */ }
   194→  }, [traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter]);
   195→
   196→  // Track latest alert ID for pulse animation on the badge
   197→  const [latestAlertId, setLatestAlertId] = useState<string | null>(null);
   198→
   199→  // WebSocket for real-time alerts
   200→  // NOTE: The alert-streamer mini-service has been upgraded to Socket.IO.
   201→  //       The raw WebSocket connection below still works because Socket.IO serves
   202→  //       a plain HTTP server that can accept ws:// upgrades. To fully migrate the
   203→  //       client to Socket.IO, replace this block with the socket.io-client library
   204→  //       and listen for 'new-alert' events from the 'alerts' room.
   205→  useEffect(() => {
   206→    let ws: WebSocket | null = null;
   207→    try {
   208→      ws = new WebSocket('ws://localhost:3001/?XTransformPort=3001');
   209→      ws.onmessage = (event) => {
   210→        try {
   211→          const data = JSON.parse(event.data);
   212→          if (data.type === 'new_alert') {
   213→            setAlertItems((prev) => [data.alert, ...prev].slice(0, 20));
   214→            setLatestAlertId(data.alert.id);
   215→          }
   216→        } catch (e) {
   217→          // ignore parse errors
   218→        }
   219→      };
   220→    } catch (e) {
   221→      // WebSocket not available, that's ok
   222→    }
   223→    return () => {
   224→      if (ws) ws.close();
   225→    };
   226→  }, []);
   227→
   228→  const fetchAgents = useCallback(async () => {
   229→    try {
   230→      const res = await fetch('/api/agents').then((r) => r.json());
   231→      setAgents(res);
   232→    } catch {
   233→      // silently fail
   234→    }
   235→  }, []);
   236→
   237→  const fetchAlerts = useCallback(async () => {
   238→    try {
   239→      const res = await fetch('/api/alerts').then((r) => r.json());
   240→      setAlertItems(res);
   241→    } catch {
   242→      // silently fail
   243→    }
   244→  }, []);
   245→
   246→  const fetchIssues = useCallback(async () => {
   247→    try {
   248→      const issuesRes = await fetch('/api/issues').then((r) => r.json());
   249→      setIssues(issuesRes);
   250→    } catch {
   251→      // silently fail
   252→    }
   253→  }, []);
   254→
   255→  // Auto-refresh: agents every 60s, alerts every 15s
   256→  useAutoRefresh(fetchAgents, 60000);
   257→  useAutoRefresh(fetchAlerts, 15000);
   258→
   259→  useEffect(() => {
   260→    async function fetchData() {
   261→      try {
   262→        const [agentsRes, tracesRes, issuesRes, alertsRes, metricsRes] =
   263→          await Promise.all([
   264→            fetch('/api/agents').then((r) => r.json()),
   265→            fetch('/api/traces').then((r) => r.json()),
   266→            fetch('/api/issues').then((r) => r.json()),
   267→            fetch('/api/alerts').then((r) => r.json()),
   268→            fetch('/api/metrics').then((r) => r.json()),
   269→          ]);
   270→        setAgents(agentsRes);
   271→        setTraces(tracesRes);
   272→        setIssues(issuesRes);
   273→        setAlertItems(alertsRes);
   274→        setMetrics(metricsRes);
   275→      } catch (e) {
   276→        console.error('Failed to fetch dashboard data:', e);
   277→      } finally {
   278→        setLoading(false);
   279→      }
   280→    }
   281→    fetchData();
   282→  }, []);
   283→
   284→  const handleCompareAgent = (agent: Agent) => {
   285→    if (comparisonIds.includes(agent.id)) {
   286→      setComparisonIds(comparisonIds.filter(id => id !== agent.id));
   287→      return;
   288→    }
   289→    const newIds = [...comparisonIds, agent.id].slice(-2);
   290→    setComparisonIds(newIds);
   291→    if (newIds.length === 2) {
   292→      const a = agents.find(a => a.id === newIds[0]);
   293→      const b = agents.find(a => a.id === newIds[1]);
   294→      if (a && b) {
   295→        setComparisonAgents([a, b]);
   296→        setComparisonIds([]);
   297→      }
   298→    }
   299→  };
   300→
   301→  const filteredTraces = traces.filter((t) => {
   302→    const matchesAgent = !selectedAgentId || t.agentId === selectedAgentId;
   303→    const matchesSearch = !traceSearch ||
   304→      t.traceId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   305→      t.agentId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   306→      t.status.toLowerCase().includes(traceSearch.toLowerCase());
   307→    const matchesStatus = traceStatusFilter === 'all' || t.status === traceStatusFilter;
   308→    return matchesAgent && matchesSearch && matchesStatus;
   309→  });
   310→
   311→  const filteredIssues = issues.filter((issue) => {
   312→    const matchesSearch = !issueSearch ||
   313→      issue.title.toLowerCase().includes(issueSearch.toLowerCase()) ||
   314→      issue.agentName.toLowerCase().includes(issueSearch.toLowerCase()) ||
   315→      issue.description.toLowerCase().includes(issueSearch.toLowerCase());
   316→    const matchesSeverity = issueSeverityFilter === 'all' || issue.severity === issueSeverityFilter;
   317→    const matchesStatus = issueStatusFilter === 'all' || issue.status === issueStatusFilter;
   318→    return matchesSearch && matchesSeverity && matchesStatus;
   319→  });
   320→
   321→  return (
   322→    <section id="dashboard" className="relative py-24 sm:py-32 bg-gray-50/50 dark:bg-gray-900/30">
   323→      <div className="absolute inset-0 bg-grid-pattern opacity-30" />
   324→      <div ref={ref} className="relative z-10 mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   325→        {/* Section header */}
   326→        <motion.div
   327→          initial={{ opacity: 0, y: 20 }}
   328→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   329→          transition={{ duration: 0.6 }}
   330→          className="text-center mb-10"
   331→        >
   332→          <div className="inline-flex items-center gap-2 rounded-full border border-red-200 dark:border-red-800/50 bg-red-50 dark:bg-red-950/30 px-3.5 py-1.5 text-xs font-medium text-[#dc2626] mb-4">
   333→            <div className="h-2 w-2 rounded-full bg-[#dc2626] animate-pulse" />
   334→            Live Demo
   335→          </div>
   336→          <h2 className="text-3xl sm:text-4xl font-bold tracking-tight text-gray-900 dark:text-gray-100">
   337→            The{'\u00A0'}
   338→            <span className="text-gradient">observability dashboard</span>
   339→          </h2>
   340→          <p className="mt-4 text-lg text-gray-500 dark:text-gray-400 max-w-2xl mx-auto">
   341→            Real-time monitoring of AI agent traces, issues, and metrics —
   342→            everything running live below.
   343→          </p>
   344→        </motion.div>
   345→
   346→        {/* Dashboard container */}
   347→        <motion.div
   348→          initial={{ opacity: 0, y: 30 }}
   349→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   350→          transition={{ duration: 0.7, delay: 0.1 }}
   351→          className="animated-border rounded-2xl bg-white dark:bg-gray-950 shadow-xl shadow-gray-200/50 dark:shadow-gray-900/50 overflow-hidden"
   352→        >
   353→          {/* Dashboard header with tabs */}
   354→          <div className="border-b border-gray-100 dark:border-gray-800 px-4 sm:px-6">
   355→            <div className="flex items-center justify-between">
   356→              <div data-tour="tabs" className="flex items-center gap-1 overflow-x-auto scrollbar-hide">
   357→                {tabs.map((tab) => {
   358→                  const Icon = tab.icon;
   359→                  const isActive = activeTab === tab.id;
   360→                  return (
   361→                    <button
   362→                      key={tab.id}
   363→                      onClick={() => handleTabChange(tab.id)}
   364→                      className={`relative flex items-center gap-1.5 px-4 py-3.5 text-sm font-medium whitespace-nowrap border-b-2 -mb-px transition-all duration-200 ${
   365→                        isActive
   366→                          ? 'text-[#dc2626] border-[#dc2626]'
   367→                          : 'text-gray-400 dark:text-gray-500 border-transparent hover:text-gray-600 dark:hover:text-gray-300'
   368→                      }`}
   369→                    >
   370→                      <Icon className="h-4 w-4" />
   371→                      {tab.label}
   372→                      {isActive && (
   373→                        <div className="absolute bottom-0 left-1/2 -translate-x-1/2 h-0.5 w-8 bg-gradient-to-r from-transparent via-[#dc2626] to-transparent rounded-full" />
   374→                      )}
   375→                      {tab.id === 'issues' && issues.length > 0 && (
   376→                        <span className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white">
   377→                          {issues.filter((i) => i.status !== 'resolved').length}
   378→                        </span>
   379→                      )}
   380→                      {tab.id === 'alerts' && alertItems.filter((a) => a.status === 'unread').length > 0 && (
   381→                        <motion.span
   382→                          key={latestAlertId ?? 'badge'}
   383→                          initial={{ scale: 1.3 }}
   384→                          animate={{ scale: 1 }}
   385→                          transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   386→                          className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white"
   387→                        >
   388→                          {alertItems.filter((a) => a.status === 'unread').length}
   389→                        </motion.span>
   390→                      )}
   391→                    </button>
   392→                  );
   393→                })}
   394→              </div>
   395→              <div className="hidden sm:flex items-center gap-2">
   396→                <div className="flex items-center gap-2 text-xs text-gray-400 dark:text-gray-500">
   397→                  <div className="h-2 w-2 rounded-full bg-emerald-500 animate-pulse" />
   398→                  Live
   399→                </div>
   400→                <button
   401→                  onClick={() => setSettingsOpen(true)}
   402→                  className="ml-2 h-7 w-7 rounded-lg flex items-center justify-center text-gray-400 dark:text-gray-500 hover:text-[#dc2626] hover:bg-red-50 dark:hover:bg-red-950/30 transition-colors focus-ring"
   403→                  aria-label="Open settings"
   404→                >
   405→                  <Settings className="h-4 w-4" />
   406→                </button>
   407→              </div>
   408→            </div>
   409→          </div>
   410→
   411→          {/* Auth-aware banner: Demo Mode or Authenticated User */}
   412→          {showBanner && (
   413→            <div className="border-b border-gray-100 dark:border-gray-800">
   414→              <div className="px-4 sm:px-6 py-2.5 flex items-center justify-between gap-3">
   415→                {isAuthenticated ? (
   416→                  <div className="flex items-center gap-2.5 min-w-0">
   417→                    <div className="h-7 w-7 rounded-full bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shrink-0">
   418→                      <User className="h-3.5 w-3.5 text-emerald-600 dark:text-emerald-400" />
   419→                    </div>
   420→                    <div className="min-w-0">
   421→                      <p className="text-xs text-gray-700 dark:text-gray-300 truncate">
   422→                        Signed in as{' '}
   423→                        <span className="font-semibold text-gray-900 dark:text-gray-100">{session.user.name || session.user.email}</span>
   424→                      </p>
   425→                      <p className="text-[11px] text-gray-400 dark:text-gray-500">{(session.user as { role?: string }).role || 'viewer'}</p>
   426→                    </div>
   427→                    <Badge variant="outline" className="ml-2 shrink-0 border-emerald-200 dark:border-emerald-800/50 text-emerald-700 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-950/30 text-[10px] px-2">
   428→                      Authenticated
   429→                    </Badge>
   430→                  </div>
   431→                ) : (
   432→                  <div className="flex items-center gap-2.5 min-w-0">
   433→                    <div className="h-7 w-7 rounded-full bg-amber-100 dark:bg-amber-900/40 flex items-center justify-center shrink-0">
   434→                      <Eye className="h-3.5 w-3.5 text-amber-600 dark:text-amber-400" />
   435→                    </div>
   436→                    <p className="text-xs text-amber-700 dark:text-amber-300 min-w-0">
   437→                      <span className="font-semibold">Demo Mode</span>{' — '}
   438→                      <span className="text-amber-600 dark:text-amber-400">Viewing simulated data</span>
   439→                    </p>
   440→                    <Badge variant="outline" className="ml-2 shrink-0 border-amber-200 dark:border-amber-800/50 text-amber-700 dark:text-amber-400 bg-amber-50 dark:bg-amber-950/30 text-[10px] px-2">
   441→                      Unauthenticated
   442→                    </Badge>
   443→                  </div>
   444→                )}
   445→                <button
   446→                  onClick={handleDismissBanner}
   447→                  className="h-6 w-6 rounded-md flex items-center justify-center text-gray-400 dark:text-gray-500 hover:text-gray-600 dark:hover:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors shrink-0 focus-ring"
   448→                  aria-label="Dismiss banner"
   449→                >
   450→                  <X className="h-3.5 w-3.5" />
   451→                </button>
   452→              </div>
   453→            </div>
   454→          )}
   455→
   456→          {/* Dashboard content */}
   457→          <div className="p-4 sm:p-6">
   458→            {loading ? (
   459→              <DashboardSkeleton />
   460→            ) : (
   461→              <>
   462→                {/* OVERVIEW TAB */}
   463→                {activeTab === 'overview' && (
   464→                  <div className="space-y-6"
   465→                  >
   466→                    {metrics?.cards && (
   467→                      <div data-tour="metrics">
   468→                        <MetricCards cards={metrics.cards} />
   469→                      </div>
   470→                    )}
   471→
   472→                    <div className="grid lg:grid-cols-3 gap-6">
   473→                      <div className="lg:col-span-2">
   474→                        <div className="flex items-center gap-2 mb-4">
   475→                          <Bot className="h-4 w-4 text-gray-400 dark:text-gray-500" />
   476→                          <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100">Monitored Agents</h3>
   477→                          <span className="text-xs text-gray-400 dark:text-gray-500">({agents.length})</span>
   478→                        </div>
   479→                        <div data-tour="agents"><AgentGrid agents={agents} onSelect={(agent) => setDetailAgent(agent)} onCompare={handleCompareAgent} comparisonIds={comparisonIds} />
   480→                        {comparisonIds.length === 1 && (
   481→                          <p className="text-xs text-[#dc2626] mt-2 flex items-center gap-1.5"><GitCompareArrows className="h-3 w-3" />1 agent selected — click another to compare</p>
   482→                        )}
   483→                        </div>
   484→                      </div>
   485→                      <div>
   486→                        <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100 mb-4">MCP Fix Workflow</h3>
   487→                        <div data-tour="mcp"><McpPanel /></div>
   488→                      </div>
   489→                    </div>
   490→
   491→                    {metrics?.timeSeries && metrics?.severityBreakdown && metrics?.frameworkDistribution && (
   492→                      <div data-tour="charts">
   493→                        <MetricsCharts
   494→                          timeSeries={metrics.timeSeries}
   495→                          severity={metrics.severityBreakdown}
   496→                          frameworks={metrics.frameworkDistribution}
   497→                        />
   498→                      </div>
   499→                    )}
   500→                  </div>
   501→                )}
   502→
   503→                {/* TRACES TAB */}
   504→                {activeTab === 'traces' && (
