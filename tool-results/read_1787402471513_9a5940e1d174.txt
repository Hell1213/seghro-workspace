   102→
   103→export function DashboardSection() {
   104→  const ref = useRef<HTMLDivElement>(null);
   105→  const isInView = useInView(ref, { once: true, margin: '-100px' });
   106→  const { data: session } = useSession();
   107→  const [activeTab, setActiveTab] = useState<TabId>('overview');
   108→
   109→  // Auth-aware banner: demo mode vs authenticated user
   110→  const isAuthenticated = !!session?.user?.email;
   111→  const [bannerDismissed, setBannerDismissed] = useState(false);
   112→
   113→  useEffect(() => {
   114→    try {
   115→      const dismissed = localStorage.getItem('sentinel-auth-banner-dismissed');
   116→      if (dismissed === 'true') setBannerDismissed(true);
   117→    } catch { /* ignore */ }
   118→  }, []);
   119→
   120→  const handleDismissBanner = useCallback(() => {
   121→    setBannerDismissed(true);
   122→    try {
   123→      localStorage.setItem('sentinel-auth-banner-dismissed', 'true');
   124→    } catch { /* ignore */ }
   125→  }, []);
   126→
   127→  const showBanner = useMemo(
   128→    () => !bannerDismissed,
   129→    [bannerDismissed]
   130→  );
   131→
   132→  // Sync tab to URL hash
   133→  useEffect(() => {
   134→    const hash = window.location.hash.replace('#dashboard-', '').replace('#', '');
   135→    if (['overview', 'traces', 'issues', 'alerts', 'api-health'].includes(hash)) {
   136→      setActiveTab(hash as TabId);
   137→    }
   138→  }, []);
   139→
   140→  const handleTabChange = (tab: TabId) => {
   141→    setActiveTab(tab);
   142→    window.history.replaceState(null, '', `#dashboard-${tab}`);
   143→  };
   144→  const [selectedAgentId, setSelectedAgentId] = useState<string | null>(null);
   145→  const [detailAgent, setDetailAgent] = useState<Agent | null>(null);
   146→  const [comparisonIds, setComparisonIds] = useState<string[]>([]);
   147→  const [comparisonAgents, setComparisonAgents] = useState<[Agent, Agent] | null>(null);
   148→  const [showWaterfall, setShowWaterfall] = useState(true);
   149→  const [settingsOpen, setSettingsOpen] = useState(false);
   150→  const [agents, setAgents] = useState<Agent[]>([]);
   151→  const [traces, setTraces] = useState<Trace[]>([]);
   152→  const [issues, setIssues] = useState<Issue[]>([]);
   153→  const [alertItems, setAlertItems] = useState<AlertItem[]>([]);
   154→  const [metrics, setMetrics] = useState<any>(null);
   155→  const [loading, setLoading] = useState(true);
   156→
   157→  // Search and filter state
   158→  const [issueSearch, setIssueSearch] = useState('');
   159→  const [issueSeverityFilter, setIssueSeverityFilter] = useState<string>('all');
   160→  const [issueStatusFilter, setIssueStatusFilter] = useState<string>('all');
   161→  const [showFilters, setShowFilters] = useState(false);
   162→
   163→  // Trace search and filter state
   164→  const [traceSearch, setTraceSearch] = useState('');
   165→  const [traceStatusFilter, setTraceStatusFilter] = useState<string>('all');
   166→
   167→  // Filter persistence via localStorage
   168→  useEffect(() => {
   169→    try {
   170→      const saved = localStorage.getItem('sentinel-dash-filters');
   171→      if (saved) {
   172→        const f = JSON.parse(saved);
   173→        if (f.traceSearch) setTraceSearch(f.traceSearch);
   174→        if (f.issueSearch) setIssueSearch(f.issueSearch);
   175→        if (f.traceStatusFilter) setTraceStatusFilter(f.traceStatusFilter);
   176→        if (f.issueSeverityFilter) setIssueSeverityFilter(f.issueSeverityFilter);
   177→        if (f.issueStatusFilter) setIssueStatusFilter(f.issueStatusFilter);
   178→      }
   179→    } catch { /* ignore */ }
   180→  }, []);
   181→
   182→  useEffect(() => {
   183→    try {
   184→      localStorage.setItem('sentinel-dash-filters', JSON.stringify({
   185→        traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter,
   186→      }));
   187→    } catch { /* ignore */ }
   188→  }, [traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter]);
   189→
   190→  // Track latest alert ID for pulse animation on the badge
   191→  const [latestAlertId, setLatestAlertId] = useState<string | null>(null);
   192→
   193→  // WebSocket for real-time alerts
   194→  // NOTE: The alert-streamer mini-service has been upgraded to Socket.IO.
   195→  //       The raw WebSocket connection below still works because Socket.IO serves
   196→  //       a plain HTTP server that can accept ws:// upgrades. To fully migrate the
   197→  //       client to Socket.IO, replace this block with the socket.io-client library
   198→  //       and listen for 'new-alert' events from the 'alerts' room.
   199→  useEffect(() => {
   200→    let ws: WebSocket | null = null;
   201→    try {
   202→      ws = new WebSocket('ws://localhost:3001/?XTransformPort=3001');
   203→      ws.onmessage = (event) => {
   204→        try {
   205→          const data = JSON.parse(event.data);
   206→          if (data.type === 'new_alert') {
   207→            setAlertItems((prev) => [data.alert, ...prev].slice(0, 20));
   208→            setLatestAlertId(data.alert.id);
   209→          }
   210→        } catch (e) {
   211→          // ignore parse errors
   212→        }
   213→      };
   214→    } catch (e) {
   215→      // WebSocket not available, that's ok
   216→    }
   217→    return () => {
   218→      if (ws) ws.close();
   219→    };
   220→  }, []);
   221→
   222→  const fetchAgents = useCallback(async () => {
   223→    try {
   224→      const res = await fetch('/api/agents').then((r) => r.json());
   225→      setAgents(res);
   226→    } catch {
   227→      // silently fail
   228→    }
   229→  }, []);
   230→
   231→  const fetchAlerts = useCallback(async () => {
   232→    try {
   233→      const res = await fetch('/api/alerts').then((r) => r.json());
   234→      setAlertItems(res);
   235→    } catch {
   236→      // silently fail
   237→    }
   238→  }, []);
   239→
   240→  const fetchIssues = useCallback(async () => {
   241→    try {
   242→      const issuesRes = await fetch('/api/issues').then((r) => r.json());
   243→      setIssues(issuesRes);
   244→    } catch {
   245→      // silently fail
   246→    }
   247→  }, []);
   248→
   249→  // Auto-refresh: agents every 60s, alerts every 15s
   250→  useAutoRefresh(fetchAgents, 60000);
   251→  useAutoRefresh(fetchAlerts, 15000);
   252→
   253→  useEffect(() => {
   254→    async function fetchData() {
   255→      try {
   256→        const [agentsRes, tracesRes, issuesRes, alertsRes, metricsRes] =
   257→          await Promise.all([
   258→            fetch('/api/agents').then((r) => r.json()),
   259→            fetch('/api/traces').then((r) => r.json()),
   260→            fetch('/api/issues').then((r) => r.json()),
   261→            fetch('/api/alerts').then((r) => r.json()),
   262→            fetch('/api/metrics').then((r) => r.json()),
   263→          ]);
   264→        setAgents(agentsRes);
   265→        setTraces(tracesRes);
   266→        setIssues(issuesRes);
   267→        setAlertItems(alertsRes);
   268→        setMetrics(metricsRes);
   269→      } catch (e) {
   270→        console.error('Failed to fetch dashboard data:', e);
   271→      } finally {
   272→        setLoading(false);
   273→      }
   274→    }
   275→    fetchData();
   276→  }, []);
   277→
   278→  const handleCompareAgent = (agent: Agent) => {
   279→    if (comparisonIds.includes(agent.id)) {
   280→      setComparisonIds(comparisonIds.filter(id => id !== agent.id));
   281→      return;
   282→    }
   283→    const newIds = [...comparisonIds, agent.id].slice(-2);
   284→    setComparisonIds(newIds);
   285→    if (newIds.length === 2) {
   286→      const a = agents.find(a => a.id === newIds[0]);
   287→      const b = agents.find(a => a.id === newIds[1]);
   288→      if (a && b) {
   289→        setComparisonAgents([a, b]);
   290→        setComparisonIds([]);
   291→      }
   292→    }
   293→  };
   294→
   295→  const filteredTraces = traces.filter((t) => {
   296→    const matchesAgent = !selectedAgentId || t.agentId === selectedAgentId;
   297→    const matchesSearch = !traceSearch ||
   298→      t.traceId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   299→      t.agentId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   300→      t.status.toLowerCase().includes(traceSearch.toLowerCase());
   301→    const matchesStatus = traceStatusFilter === 'all' || t.status === traceStatusFilter;
   302→    return matchesAgent && matchesSearch && matchesStatus;
   303→  });
   304→
   305→  const filteredIssues = issues.filter((issue) => {
   306→    const matchesSearch = !issueSearch ||
   307→      issue.title.toLowerCase().includes(issueSearch.toLowerCase()) ||
   308→      issue.agentName.toLowerCase().includes(issueSearch.toLowerCase()) ||
   309→      issue.description.toLowerCase().includes(issueSearch.toLowerCase());
   310→    const matchesSeverity = issueSeverityFilter === 'all' || issue.severity === issueSeverityFilter;
   311→    const matchesStatus = issueStatusFilter === 'all' || issue.status === issueStatusFilter;
   312→    return matchesSearch && matchesSeverity && matchesStatus;
   313→  });
   314→
   315→  return (
   316→    <section id="dashboard" className="relative py-24 sm:py-32 bg-gray-50/50 dark:bg-gray-900/30">
   317→      <div className="absolute inset-0 bg-grid-pattern opacity-30" />
   318→      <div ref={ref} className="relative z-10 mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   319→        {/* Section header */}
   320→        <motion.div
   321→          initial={{ opacity: 0, y: 20 }}
   322→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   323→          transition={{ duration: 0.6 }}
   324→          className="text-center mb-10"
   325→        >
   326→          <div className="inline-flex items-center gap-2 rounded-full border border-red-200 dark:border-red-800/50 bg-red-50 dark:bg-red-950/30 px-3.5 py-1.5 text-xs font-medium text-[#dc2626] mb-4">
   327→            <div className="h-2 w-2 rounded-full bg-[#dc2626] animate-pulse" />
   328→            Live Demo
   329→          </div>
   330→          <h2 className="text-3xl sm:text-4xl font-bold tracking-tight text-gray-900 dark:text-gray-100">
   331→            The{'\u00A0'}
   332→            <span className="text-gradient">observability dashboard</span>
   333→          </h2>
   334→          <p className="mt-4 text-lg text-gray-500 dark:text-gray-400 max-w-2xl mx-auto">
   335→            Real-time monitoring of AI agent traces, issues, and metrics —
   336→            everything running live below.
   337→          </p>
   338→        </motion.div>
   339→
   340→        {/* Dashboard container */}
   341→        <motion.div
   342→          initial={{ opacity: 0, y: 30 }}
   343→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   344→          transition={{ duration: 0.7, delay: 0.1 }}
   345→          className="animated-border rounded-2xl bg-white dark:bg-gray-950 shadow-xl shadow-gray-200/50 dark:shadow-gray-900/50 overflow-hidden"
   346→        >
   347→          {/* Dashboard header with tabs */}
   348→          <div className="border-b border-gray-100 dark:border-gray-800 px-4 sm:px-6">
   349→            <div className="flex items-center justify-between">
   350→              <div data-tour="tabs" className="flex items-center gap-1 overflow-x-auto scrollbar-hide">
   351→                {tabs.map((tab) => {
   352→                  const Icon = tab.icon;
   353→                  const isActive = activeTab === tab.id;
   354→                  return (
   355→                    <button
   356→                      key={tab.id}
   357→                      onClick={() => handleTabChange(tab.id)}
   358→                      className={`relative flex items-center gap-1.5 px-4 py-3.5 text-sm font-medium whitespace-nowrap border-b-2 -mb-px transition-all duration-200 ${
   359→                        isActive
   360→                          ? 'text-[#dc2626] border-[#dc2626]'
   361→                          : 'text-gray-400 dark:text-gray-500 border-transparent hover:text-gray-600 dark:hover:text-gray-300'
   362→                      }`}
   363→                    >
   364→                      <Icon className="h-4 w-4" />
   365→                      {tab.label}
   366→                      {isActive && (
   367→                        <div className="absolute bottom-0 left-1/2 -translate-x-1/2 h-0.5 w-8 bg-gradient-to-r from-transparent via-[#dc2626] to-transparent rounded-full" />
   368→                      )}
   369→                      {tab.id === 'issues' && issues.length > 0 && (
   370→                        <span className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white">
   371→                          {issues.filter((i) => i.status !== 'resolved').length}
   372→                        </span>
   373→                      )}
   374→                      {tab.id === 'alerts' && alertItems.filter((a) => a.status === 'unread').length > 0 && (
   375→                        <motion.span
   376→                          key={latestAlertId ?? 'badge'}
   377→                          initial={{ scale: 1.3 }}
   378→                          animate={{ scale: 1 }}
   379→                          transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   380→                          className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white"
   381→                        >
   382→                          {alertItems.filter((a) => a.status === 'unread').length}
   383→                        </motion.span>
   384→                      )}
   385→                    </button>
   386→                  );
   387→                })}
   388→              </div>
   389→              <div className="hidden sm:flex items-center gap-2">
   390→                <div className="flex items-center gap-2 text-xs text-gray-400 dark:text-gray-500">
   391→                  <div className="h-2 w-2 rounded-full bg-emerald-500 animate-pulse" />
   392→                  Live
   393→                </div>
   394→                <button
   395→                  onClick={() => setSettingsOpen(true)}
   396→                  className="ml-2 h-7 w-7 rounded-lg flex items-center justify-center text-gray-400 dark:text-gray-500 hover:text-[#dc2626] hover:bg-red-50 dark:hover:bg-red-950/30 transition-colors focus-ring"
   397→                  aria-label="Open settings"
   398→                >
   399→                  <Settings className="h-4 w-4" />
   400→                </button>
   401→              </div>
   402→            </div>
   403→          </div>
   404→
   405→          {/* Auth-aware banner: Demo Mode or Authenticated User */}
   406→          {showBanner && (
   407→            <div className="border-b border-gray-100 dark:border-gray-800">
   408→              <div className="px-4 sm:px-6 py-2.5 flex items-center justify-between gap-3">
   409→                {isAuthenticated ? (
   410→                  <div className="flex items-center gap-2.5 min-w-0">
   411→                    <div className="h-7 w-7 rounded-full bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shrink-0">
   412→                      <User className="h-3.5 w-3.5 text-emerald-600 dark:text-emerald-400" />
   413→                    </div>
   414→                    <div className="min-w-0">
   415→                      <p className="text-xs text-gray-700 dark:text-gray-300 truncate">
   416→                        Signed in as{' '}
   417→                        <span className="font-semibold text-gray-900 dark:text-gray-100">{session.user.name || session.user.email}</span>
   418→                      </p>
   419→                      <p className="text-[11px] text-gray-400 dark:text-gray-500">{(session.user as { role?: string }).role || 'viewer'}</p>
   420→                    </div>
   421→                    <Badge variant="outline" className="ml-2 shrink-0 border-emerald-200 dark:border-emerald-800/50 text-emerald-700 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-950/30 text-[10px] px-2">
   422→                      Authenticated
   423→                    </Badge>
   424→                  </div>
   425→                ) : (
   426→                  <div className="flex items-center gap-2.5 min-w-0">
   427→                    <div className="h-7 w-7 rounded-full bg-amber-100 dark:bg-amber-900/40 flex items-center justify-center shrink-0">
   428→                      <Eye className="h-3.5 w-3.5 text-amber-600 dark:text-amber-400" />
   429→                    </div>
   430→                    <p className="text-xs text-amber-700 dark:text-amber-300 min-w-0">
   431→                      <span className="font-semibold">Demo Mode</span>{' — '}
   432→                      <span className="text-amber-600 dark:text-amber-400">Viewing simulated data</span>
   433→                    </p>
   434→                    <Badge variant="outline" className="ml-2 shrink-0 border-amber-200 dark:border-amber-800/50 text-amber-700 dark:text-amber-400 bg-amber-50 dark:bg-amber-950/30 text-[10px] px-2">
   435→                      Unauthenticated
   436→                    </Badge>
   437→                  </div>
   438→                )}
   439→                <button
   440→                  onClick={handleDismissBanner}
   441→                  className="h-6 w-6 rounded-md flex items-center justify-center text-gray-400 dark:text-gray-500 hover:text-gray-600 dark:hover:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors shrink-0 focus-ring"
   442→                  aria-label="Dismiss banner"
   443→                >
   444→                  <X className="h-3.5 w-3.5" />
   445→                </button>
   446→              </div>
   447→            </div>
   448→          )}
   449→
   450→          {/* Dashboard content */}
   451→          <div className="p-4 sm:p-6">
   452→            {loading ? (
   453→              <DashboardSkeleton />
   454→            ) : (
   455→              <>
   456→                {/* OVERVIEW TAB */}
   457→                {activeTab === 'overview' && (
   458→                  <div className="space-y-6"
   459→                  >
   460→                    {metrics?.cards && (
   461→                      <div data-tour="metrics">
   462→                        <MetricCards cards={metrics.cards} />
   463→                      </div>
   464→                    )}
   465→
   466→                    <div className="grid lg:grid-cols-3 gap-6">
   467→                      <div className="lg:col-span-2">
   468→                        <div className="flex items-center gap-2 mb-4">
   469→                          <Bot className="h-4 w-4 text-gray-400 dark:text-gray-500" />
   470→                          <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100">Monitored Agents</h3>
   471→                          <span className="text-xs text-gray-400 dark:text-gray-500">({agents.length})</span>
   472→                        </div>
   473→                        <div data-tour="agents"><AgentGrid agents={agents} onSelect={(agent) => setDetailAgent(agent)} onCompare={handleCompareAgent} comparisonIds={comparisonIds} />
   474→                        {comparisonIds.length === 1 && (
   475→                          <p className="text-xs text-[#dc2626] mt-2 flex items-center gap-1.5"><GitCompareArrows className="h-3 w-3" />1 agent selected — click another to compare</p>
   476→                        )}
   477→                        </div>
   478→                      </div>
   479→                      <div>
   480→                        <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100 mb-4">MCP Fix Workflow</h3>
   481→                        <div data-tour="mcp"><McpPanel /></div>
   482→                      </div>
   483→                    </div>
   484→
   485→                    {metrics?.timeSeries && metrics?.severityBreakdown && metrics?.frameworkDistribution && (
   486→                      <div data-tour="charts">
   487→                        <MetricsCharts
   488→                          timeSeries={metrics.timeSeries}
   489→                          severity={metrics.severityBreakdown}
   490→                          frameworks={metrics.frameworkDistribution}
   491→                        />
   492→                      </div>
   493→                    )}
   494→                  </div>
   495→                )}
   496→
   497→                {/* TRACES TAB */}
   498→                {activeTab === 'traces' && (
   499→                  <div>
   500→                    <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 mb-4">
   501→                      <div className="flex items-center gap-2">
