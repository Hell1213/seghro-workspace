   103→   103→export function DashboardSection() {
   104→   104→  const ref = useRef<HTMLDivElement>(null);
   105→   105→  const isInView = useInView(ref, { once: true, margin: '-100px' });
   106→   106→  const { data: session } = useSession();
   107→   107→  const [activeTab, setActiveTab] = useState<TabId>('overview');
   108→   108→
   109→   109→  // Auth-aware banner: demo mode vs authenticated user
   110→   110→  const isAuthenticated = !!session?.user?.email;
   111→   111→  const [bannerDismissed, setBannerDismissed] = useState(false);
   112→   112→
   113→   113→  useEffect(() => {
   114→   114→    try {
   115→   115→      const dismissed = localStorage.getItem('sentinel-auth-banner-dismissed');
   116→   116→      if (dismissed === 'true') setBannerDismissed(true);
   117→   117→    } catch { /* ignore */ }
   118→   118→  }, []);
   119→   119→
   120→   120→  const handleDismissBanner = useCallback(() => {
   121→   121→    setBannerDismissed(true);
   122→   122→    try {
   123→   123→      localStorage.setItem('sentinel-auth-banner-dismissed', 'true');
   124→   124→    } catch { /* ignore */ }
   125→   125→  }, []);
   126→   126→
   127→   127→  const showBanner = useMemo(
   128→   128→    () => !bannerDismissed,
   129→   129→    [bannerDismissed]
   130→   130→  );
   131→   131→
   132→   132→  // Sync tab to URL hash
   133→   133→  useEffect(() => {
   134→   134→    const hash = window.location.hash.replace('#dashboard-', '').replace('#', '');
   135→   135→    if (['overview', 'traces', 'issues', 'alerts', 'api-health'].includes(hash)) {
   136→   136→      setActiveTab(hash as TabId);
   137→   137→    }
   138→   138→  }, []);
   139→   139→
   140→   140→  const handleTabChange = (tab: TabId) => {
   141→   141→    setActiveTab(tab);
   142→   142→    window.history.replaceState(null, '', `#dashboard-${tab}`);
   143→   143→  };
   144→   144→  const [selectedAgentId, setSelectedAgentId] = useState<string | null>(null);
   145→   145→  const [detailAgent, setDetailAgent] = useState<Agent | null>(null);
   146→   146→  const [comparisonIds, setComparisonIds] = useState<string[]>([]);
   147→   147→  const [comparisonAgents, setComparisonAgents] = useState<[Agent, Agent] | null>(null);
   148→   148→  const [showWaterfall, setShowWaterfall] = useState(true);
   149→   149→  const [settingsOpen, setSettingsOpen] = useState(false);
   150→   150→  const [agents, setAgents] = useState<Agent[]>([]);
   151→   151→  const [traces, setTraces] = useState<Trace[]>([]);
   152→   152→  const [issues, setIssues] = useState<Issue[]>([]);
   153→   153→  const [alertItems, setAlertItems] = useState<AlertItem[]>([]);
   154→   154→  const [metrics, setMetrics] = useState<any>(null);
   155→   155→  const [loading, setLoading] = useState(true);
   156→   156→
   157→   157→  // Search and filter state
   158→   158→  const [issueSearch, setIssueSearch] = useState('');
   159→   159→  const [issueSeverityFilter, setIssueSeverityFilter] = useState<string>('all');
   160→   160→  const [issueStatusFilter, setIssueStatusFilter] = useState<string>('all');
   161→   161→  const [showFilters, setShowFilters] = useState(false);
   162→   162→
   163→   163→  // Trace search and filter state
   164→   164→  const [traceSearch, setTraceSearch] = useState('');
   165→   165→  const [traceStatusFilter, setTraceStatusFilter] = useState<string>('all');
   166→   166→
   167→   167→  // Filter persistence via localStorage
   168→   168→  useEffect(() => {
   169→   169→    try {
   170→   170→      const saved = localStorage.getItem('sentinel-dash-filters');
   171→   171→      if (saved) {
   172→   172→        const f = JSON.parse(saved);
   173→   173→        if (f.traceSearch) setTraceSearch(f.traceSearch);
   174→   174→        if (f.issueSearch) setIssueSearch(f.issueSearch);
   175→   175→        if (f.traceStatusFilter) setTraceStatusFilter(f.traceStatusFilter);
   176→   176→        if (f.issueSeverityFilter) setIssueSeverityFilter(f.issueSeverityFilter);
   177→   177→        if (f.issueStatusFilter) setIssueStatusFilter(f.issueStatusFilter);
   178→   178→      }
   179→   179→    } catch { /* ignore */ }
   180→   180→  }, []);
   181→   181→
   182→   182→  useEffect(() => {
   183→   183→    try {
   184→   184→      localStorage.setItem('sentinel-dash-filters', JSON.stringify({
   185→   185→        traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter,
   186→   186→      }));
   187→   187→    } catch { /* ignore */ }
   188→   188→  }, [traceSearch, issueSearch, traceStatusFilter, issueSeverityFilter, issueStatusFilter]);
   189→   189→
   190→   190→  // Track latest alert ID for pulse animation on the badge
   191→   191→  const [latestAlertId, setLatestAlertId] = useState<string | null>(null);
   192→   192→
   193→   193→  // WebSocket for real-time alerts
   194→   194→  // NOTE: The alert-streamer mini-service has been upgraded to Socket.IO.
   195→   195→  //       The raw WebSocket connection below still works because Socket.IO serves
   196→   196→  //       a plain HTTP server that can accept ws:// upgrades. To fully migrate the
   197→   197→  //       client to Socket.IO, replace this block with the socket.io-client library
   198→   198→  //       and listen for 'new-alert' events from the 'alerts' room.
   199→   199→  useEffect(() => {
   200→   200→    let ws: WebSocket | null = null;
   201→   201→    try {
   202→   202→      ws = new WebSocket('ws://localhost:3001/?XTransformPort=3001');
   203→   203→      ws.onmessage = (event) => {
   204→   204→        try {
   205→   205→          const data = JSON.parse(event.data);
   206→   206→          if (data.type === 'new_alert') {
   207→   207→            setAlertItems((prev) => [data.alert, ...prev].slice(0, 20));
   208→   208→            setLatestAlertId(data.alert.id);
   209→   209→          }
   210→   210→        } catch (e) {
   211→   211→          // ignore parse errors
   212→   212→        }
   213→   213→      };
   214→   214→    } catch (e) {
   215→   215→      // WebSocket not available, that's ok
   216→   216→    }
   217→   217→    return () => {
   218→   218→      if (ws) ws.close();
   219→   219→    };
   220→   220→  }, []);
   221→   221→
   222→   222→  const fetchAgents = useCallback(async () => {
   223→   223→    try {
   224→   224→      const res = await fetch('/api/agents').then((r) => r.json());
   225→   225→      setAgents(res);
   226→   226→    } catch {
   227→   227→      // silently fail
   228→   228→    }
   229→   229→  }, []);
   230→   230→
   231→   231→  const fetchAlerts = useCallback(async () => {
   232→   232→    try {
   233→   233→      const res = await fetch('/api/alerts').then((r) => r.json());
   234→   234→      setAlertItems(res);
   235→   235→    } catch {
   236→   236→      // silently fail
   237→   237→    }
   238→   238→  }, []);
   239→   239→
   240→   240→  const fetchIssues = useCallback(async () => {
   241→   241→    try {
   242→   242→      const issuesRes = await fetch('/api/issues').then((r) => r.json());
   243→   243→      setIssues(issuesRes);
   244→   244→    } catch {
   245→   245→      // silently fail
   246→   246→    }
   247→   247→  }, []);
   248→   248→
   249→   249→  // Auto-refresh: agents every 60s, alerts every 15s
   250→   250→  useAutoRefresh(fetchAgents, 60000);
   251→   251→  useAutoRefresh(fetchAlerts, 15000);
   252→   252→
   253→   253→  useEffect(() => {
   254→   254→    async function fetchData() {
   255→   255→      try {
   256→   256→        const [agentsRes, tracesRes, issuesRes, alertsRes, metricsRes] =
   257→   257→          await Promise.all([
   258→   258→            fetch('/api/agents').then((r) => r.json()),
   259→   259→            fetch('/api/traces').then((r) => r.json()),
   260→   260→            fetch('/api/issues').then((r) => r.json()),
   261→   261→            fetch('/api/alerts').then((r) => r.json()),
   262→   262→            fetch('/api/metrics').then((r) => r.json()),
   263→   263→          ]);
   264→   264→        setAgents(agentsRes);
   265→   265→        setTraces(tracesRes);
   266→   266→        setIssues(issuesRes);
   267→   267→        setAlertItems(alertsRes);
   268→   268→        setMetrics(metricsRes);
   269→   269→      } catch (e) {
   270→   270→        console.error('Failed to fetch dashboard data:', e);
   271→   271→      } finally {
   272→   272→        setLoading(false);
   273→   273→      }
   274→   274→    }
   275→   275→    fetchData();
   276→   276→  }, []);
   277→   277→
   278→   278→  const handleCompareAgent = (agent: Agent) => {
   279→   279→    if (comparisonIds.includes(agent.id)) {
   280→   280→      setComparisonIds(comparisonIds.filter(id => id !== agent.id));
   281→   281→      return;
   282→   282→    }
   283→   283→    const newIds = [...comparisonIds, agent.id].slice(-2);
   284→   284→    setComparisonIds(newIds);
   285→   285→    if (newIds.length === 2) {
   286→   286→      const a = agents.find(a => a.id === newIds[0]);
   287→   287→      const b = agents.find(a => a.id === newIds[1]);
   288→   288→      if (a && b) {
   289→   289→        setComparisonAgents([a, b]);
   290→   290→        setComparisonIds([]);
   291→   291→      }
   292→   292→    }
   293→   293→  };
   294→   294→
   295→   295→  const filteredTraces = traces.filter((t) => {
   296→   296→    const matchesAgent = !selectedAgentId || t.agentId === selectedAgentId;
   297→   297→    const matchesSearch = !traceSearch ||
   298→   298→      t.traceId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   299→   299→      t.agentId.toLowerCase().includes(traceSearch.toLowerCase()) ||
   300→   300→      t.status.toLowerCase().includes(traceSearch.toLowerCase());
   301→   301→    const matchesStatus = traceStatusFilter === 'all' || t.status === traceStatusFilter;
   302→   302→    return matchesAgent && matchesSearch && matchesStatus;
   303→   303→  });
   304→   304→
   305→   305→  const filteredIssues = issues.filter((issue) => {
   306→   306→    const matchesSearch = !issueSearch ||
   307→   307→      issue.title.toLowerCase().includes(issueSearch.toLowerCase()) ||
   308→   308→      issue.agentName.toLowerCase().includes(issueSearch.toLowerCase()) ||
   309→   309→      issue.description.toLowerCase().includes(issueSearch.toLowerCase());
   310→   310→    const matchesSeverity = issueSeverityFilter === 'all' || issue.severity === issueSeverityFilter;
   311→   311→    const matchesStatus = issueStatusFilter === 'all' || issue.status === issueStatusFilter;
   312→   312→    return matchesSearch && matchesSeverity && matchesStatus;
   313→   313→  });
   314→   314→
   315→   315→  return (
   316→   316→    <section id="dashboard" className="relative py-24 sm:py-32 bg-gray-50/50 dark:bg-gray-900/30">
   317→   317→      <div className="absolute inset-0 bg-grid-pattern opacity-30" />
   318→   318→      <div ref={ref} className="relative z-10 mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   319→   319→        {/* Section header */}
   320→   320→        <motion.div
   321→   321→          initial={{ opacity: 0, y: 20 }}
   322→   322→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   323→   323→          transition={{ duration: 0.6 }}
   324→   324→          className="text-center mb-10"
   325→   325→        >
   326→   326→          <div className="inline-flex items-center gap-2 rounded-full border border-red-200 dark:border-red-800/50 bg-red-50 dark:bg-red-950/30 px-3.5 py-1.5 text-xs font-medium text-[#dc2626] mb-4">
   327→   327→            <div className="h-2 w-2 rounded-full bg-[#dc2626] animate-pulse" />
   328→   328→            Live Demo
   329→   329→          </div>
   330→   330→          <h2 className="text-3xl sm:text-4xl font-bold tracking-tight text-gray-900 dark:text-gray-100">
   331→   331→            The{'\u00A0'}
   332→   332→            <span className="text-gradient">observability dashboard</span>
   333→   333→          </h2>
   334→   334→          <p className="mt-4 text-lg text-gray-500 dark:text-gray-400 max-w-2xl mx-auto">
   335→   335→            Real-time monitoring of AI agent traces, issues, and metrics —
   336→   336→            everything running live below.
   337→   337→          </p>
   338→   338→        </motion.div>
   339→   339→
   340→   340→        {/* Dashboard container */}
   341→   341→        <motion.div
   342→   342→          initial={{ opacity: 0, y: 30 }}
   343→   343→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   344→   344→          transition={{ duration: 0.7, delay: 0.1 }}
   345→   345→          className="animated-border rounded-2xl bg-white dark:bg-gray-950 shadow-xl shadow-gray-200/50 dark:shadow-gray-900/50 overflow-hidden"
   346→   346→        >
   347→   347→          {/* Dashboard header with tabs */}
   348→   348→          <div className="border-b border-gray-100 dark:border-gray-800 px-4 sm:px-6">
   349→   349→            <div className="flex items-center justify-between">
   350→   350→              <div data-tour="tabs" className="flex items-center gap-1 overflow-x-auto scrollbar-hide">
   351→   351→                {tabs.map((tab) => {
   352→   352→                  const Icon = tab.icon;
   353→   353→                  const isActive = activeTab === tab.id;
   354→   354→                  return (
   355→   355→                    <button
   356→   356→                      key={tab.id}
   357→   357→                      onClick={() => handleTabChange(tab.id)}
   358→   358→                      className={`relative flex items-center gap-1.5 px-4 py-3.5 text-sm font-medium whitespace-nowrap border-b-2 -mb-px transition-all duration-200 ${
   359→   359→                        isActive
   360→   360→                          ? 'text-[#dc2626] border-[#dc2626]'
   361→   361→                          : 'text-gray-400 dark:text-gray-500 border-transparent hover:text-gray-600 dark:hover:text-gray-300'
   362→   362→                      }`}
   363→   363→                    >
   364→   364→                      <Icon className="h-4 w-4" />
   365→   365→                      {tab.label}
   366→   366→                      {isActive && (
   367→   367→                        <div className="absolute bottom-0 left-1/2 -translate-x-1/2 h-0.5 w-8 bg-gradient-to-r from-transparent via-[#dc2626] to-transparent rounded-full" />
   368→   368→                      )}
   369→   369→                      {tab.id === 'issues' && issues.length > 0 && (
   370→   370→                        <span className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white">
   371→   371→                          {issues.filter((i) => i.status !== 'resolved').length}
   372→   372→                        </span>
   373→   373→                      )}
   374→   374→                      {tab.id === 'alerts' && alertItems.filter((a) => a.status === 'unread').length > 0 && (
   375→   375→                        <motion.span
   376→   376→                          key={latestAlertId ?? 'badge'}
   377→   377→                          initial={{ scale: 1.3 }}
   378→   378→                          animate={{ scale: 1 }}
   379→   379→                          transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   380→   380→                          className="badge-pulse flex h-4 min-w-4 items-center justify-center rounded-full bg-[#dc2626] px-1 text-[9px] font-bold text-white"
   381→   381→                        >
   382→   382→                          {alertItems.filter((a) => a.status === 'unread').length}
   383→   383→                        </motion.span>
   384→   384→                      )}
   385→   385→                    </button>
   386→   386→                  );
   387→   387→                })}
   388→   388→              </div>
   389→   389→              <div className="hidden sm:flex items-center gap-2">
   390→   390→                <div className="flex items-center gap-2 text-xs text-gray-400 dark:text-gray-500">
   391→   391→                  <div className="h-2 w-2 rounded-full bg-emerald-500 animate-pulse" />
   392→   392→                  Live
   393→   393→                </div>
   394→   394→                <button
   395→   395→                  onClick={() => setSettingsOpen(true)}
   396→   396→                  className="ml-2 h-7 w-7 rounded-lg flex items-center justify-center text-gray-400 dark:text-gray-500 hover:text-[#dc2626] hover:bg-red-50 dark:hover:bg-red-950/30 transition-colors focus-ring"
   397→   397→                  aria-label="Open settings"
   398→   398→                >
   399→   399→                  <Settings className="h-4 w-4" />
   400→   400→                </button>
   401→   401→              </div>
   402→   402→            </div>
   403→   403→          </div>
   404→   404→
   405→   405→          {/* Auth-aware banner: Demo Mode or Authenticated User */}
   406→   406→          {showBanner && (
   407→   407→            <div className="border-b border-gray-100 dark:border-gray-800">
   408→   408→              <div className="px-4 sm:px-6 py-2.5 flex items-center justify-between gap-3">
   409→   409→                {isAuthenticated ? (
   410→   410→                  <div className="flex items-center gap-2.5 min-w-0">
   411→   411→                    <div className="h-7 w-7 rounded-full bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shrink-0">
   412→   412→                      <User className="h-3.5 w-3.5 text-emerald-600 dark:text-emerald-400" />
   413→   413→                    </div>
   414→   414→                    <div className="min-w-0">
   415→   415→                      <p className="text-xs text-gray-700 dark:text-gray-300 truncate">
   416→   416→                        Signed in as{' '}
   417→   417→                        <span className="font-semibold text-gray-900 dark:text-gray-100">{session.user.name || session.user.email}</span>
   418→   418→                      </p>
   419→   419→                      <p className="text-[11px] text-gray-400 dark:text-gray-500">{(session.user as { role?: string }).role || 'viewer'}</p>
   420→   420→                    </div>
   421→   421→                    <Badge variant="outline" className="ml-2 shrink-0 border-emerald-200 dark:border-emerald-800/50 text-emerald-700 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-950/30 text-[10px] px-2">
   422→   422→                      Authenticated
   423→   423→                    </Badge>
   424→   424→                  </div>
   425→   425→                ) : (
   426→   426→                  <div className="flex items-center gap-2.5 min-w-0">
   427→   427→                    <div className="h-7 w-7 rounded-full bg-amber-100 dark:bg-amber-900/40 flex items-center justify-center shrink-0">
